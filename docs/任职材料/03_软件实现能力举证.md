复算命令，在仓库根目录运行：

```powershell
$tracked = @(git ls-files services)
$rows = foreach ($module in @('agent-gateway', 'policy-center', 'demo-business-agent')) {
    foreach ($kind in @('main', 'test')) {
        $files = @($tracked | Where-Object { $_ -like "services/$module/src/$kind/java/*.java" })
        $physical = 0
        $nonblank = 0
        foreach ($file in $files) {
            $content = @(Get-Content -LiteralPath $file -Encoding UTF8)
            $physical += $content.Count
            $nonblank += @($content | Where-Object { $_.Trim().Length -gt 0 }).Count
        }
        [pscustomobject]@{ Module=$module; Kind=$kind; Files=$files.Count; Physical=$physical; Nonblank=$nonblank }
    }
}
$rows | Format-Table -AutoSize
```
